````text
-- ============================================================
-- SQL — ЛОГИКА + СИНТАКСИС
-- ============================================================


-- ============================================================
-- 1. ОБИКНОВЕН SELECT
-- ============================================================

-- Какво искам да покажа?
--        ↓
-- От коя таблица?
--        ↓
-- Има ли условие?

SELECT Name, Grade
FROM Students;


-- ============================================================
-- 2. WHERE — ФИЛТРИРАМ РЕДОВЕ
-- ============================================================

-- Мислене:
-- "Кои конкретни редове искам?"

SELECT Name, Grade
FROM Students
WHERE Grade > 50;


-- ВАЖНО:
-- WHERE → редове
-- Не използвам WHERE за резултат от SUM/COUNT/AVG...


-- ============================================================
-- 3. ORDER BY — СОРТИРАМ
-- ============================================================

-- Филтрирам → после сортирам

SELECT Name, Grade
FROM Students
WHERE Grade > 50
ORDER BY Grade DESC;


-- ASC  → възходящо
-- DESC → низходящо


-- ============================================================
-- 4. TOP — ВЗИМАМ ОПРЕДЕЛЕН БРОЙ
-- ============================================================

-- Сортирам → взимам първите N

SELECT TOP 3 Name, Grade
FROM Students
ORDER BY Grade DESC;


-- ============================================================
-- 5. АГРЕГАТНИ ФУНКЦИИ
-- ============================================================

-- SUM  → обща сума
-- COUNT → брой
-- AVG  → средна стойност
-- MIN  → най-малка
-- MAX  → най-голяма

SELECT SUM(Price)
FROM Orders;

SELECT COUNT(*)
FROM Orders;

SELECT AVG(Grade)
FROM Students;

SELECT MIN(Grade)
FROM Students;

SELECT MAX(Grade)
FROM Students;


-- ============================================================
-- 6. GROUP BY
-- ============================================================

-- Мислене:
-- "Искам отделен резултат ЗА ВСЯКА ГРУПА."

-- групирам по клиент
-- → броя поръчките на всеки клиент

SELECT Customer, COUNT(*)
FROM Orders
GROUP BY Customer;


-- Модел:
--
-- GROUP BY → КОЙ?
-- COUNT/SUM/AVG... → КАКВО ИЗЧИСЛЯВАМ ЗА НЕГО?


-- ============================================================
-- 7. GROUP BY + SUM
-- ============================================================

-- групирам по клиент
-- → събирам цените
-- → показвам клиент + сума

SELECT Customer, SUM(Price) AS TotalPrice
FROM Orders
GROUP BY Customer;


-- ============================================================
-- 8. HAVING
-- ============================================================

-- Мислене:
-- "Искам да филтрирам ВЕЧЕ СЪЗДАДЕНИТЕ ГРУПИ."

SELECT Customer, SUM(Price) AS TotalPrice
FROM Orders
GROUP BY Customer
HAVING SUM(Price) > 500;


-- ЗАПОМНИ:
--
-- WHERE
-- → филтрира РЕДОВЕ
--
-- GROUP BY
-- → създава ГРУПИ
--
-- HAVING
-- → филтрира ГРУПИ


-- ============================================================
-- 9. WHERE + GROUP BY + HAVING
-- ============================================================

-- Първо:
-- филтрирам отделни поръчки
--
-- после:
-- групирам останалите
--
-- после:
-- филтрирам групите

SELECT Customer, SUM(Price) AS TotalPrice
FROM Orders
WHERE Price > 100
GROUP BY Customer
HAVING SUM(Price) > 500;


-- ЛОГИКА:
--
-- WHERE  → кои редове влизат?
-- GROUP BY → как ги групирам?
-- SUM → какво изчислявам?
-- HAVING → кои групи остават?


-- ============================================================
-- 10. JOIN
-- ============================================================

-- Мислене:
-- "Информацията, която ми трябва, е в две таблици."
--
-- Намирам общото поле.
--
-- Customers.Id
--       ↓
-- Orders.CustomerId

SELECT c.Name, o.Price
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId;


-- JOIN
-- → свързвам таблиците
--
-- ON
-- → казвам КАК са свързани


-- ============================================================
-- 11. LEFT JOIN
-- ============================================================

-- Мислене:
-- "Искам ВСИЧКИ клиенти,
-- дори тези без поръчки."

SELECT c.Name, o.Price
FROM Customers AS c
LEFT JOIN Orders AS o
    ON c.Id = o.CustomerId;


-- LEFT JOIN
-- → запазва всички редове от ЛЯВАТА таблица.
--
-- Винаги първо се питай:
--
-- "Коя таблица искам да запазя изцяло?"


-- ============================================================
-- 12. JOIN + GROUP BY + SUM
-- ============================================================

-- Мислене:
--
-- свързвам
-- → групирам по клиент
-- → събирам поръчките
-- → показвам клиент + сума

SELECT c.Name, SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name;


-- ============================================================
-- 13. JOIN + GROUP BY + HAVING
-- ============================================================

-- Мислене:
--
-- свързвам
-- → групирам
-- → SUM
-- → филтрирам групите

SELECT c.Name, SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name
HAVING SUM(o.Price) > 500;


-- ============================================================
-- 14. COUNT + HAVING
-- ============================================================

-- "Клиенти с поне 2 поръчки"

SELECT c.Name, COUNT(o.Id) AS OrderCount
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name
HAVING COUNT(o.Id) >= 2;


-- ============================================================
-- 15. JOIN + COUNT + SUM + HAVING
-- ============================================================

-- Мислене:
--
-- връзка
-- → групирам клиент
-- → броя поръчки
-- → сумирам поръчките
-- → филтрирам групите

SELECT
    c.Name,
    COUNT(o.Id) AS OrderCount,
    SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name
HAVING COUNT(o.Id) > 1
   AND SUM(o.Price) > 500;


-- ============================================================
-- 16. SUBQUERY — ЕДНА СТОЙНОСТ
-- ============================================================

-- Мислене:
--
-- "Първо трябва да получа една стойност,
-- после да я използвам във външния SELECT."

-- Например:
-- студенти над средната оценка

SELECT Name, Grade
FROM Students
WHERE Grade > (
    SELECT AVG(Grade)
    FROM Students
);


-- Вътрешният SELECT:
-- → намира AVG
--
-- Външният SELECT:
-- → използва AVG за сравнение


-- ============================================================
-- 17. MAX / MIN С SUBQUERY
-- ============================================================

-- Най-висока оценка

SELECT Name, Grade
FROM Students
WHERE Grade = (
    SELECT MAX(Grade)
    FROM Students
);


-- Най-ниска оценка

SELECT Name, Grade
FROM Students
WHERE Grade = (
    SELECT MIN(Grade)
    FROM Students
);


-- Мислене:
--
-- MAX/MIN → получавам ЕДНА стойност
-- → сравнявам редовете с нея


-- ============================================================
-- 18. SUBQUERY С КОНКРЕТЕН ЧОВЕК
-- ============================================================

-- "Студенти с по-висока оценка от Maria"

SELECT Name, Grade
FROM Students
WHERE Grade > (
    SELECT Grade
    FROM Students
    WHERE Name = 'Maria'
);


-- Вътрешен SELECT:
-- → намирам Grade на Maria
--
-- Външен SELECT:
-- → сравнявам всички студенти с тази стойност


-- ============================================================
-- 19. SUBQUERY + УСЛОВИЕ
-- ============================================================

-- "По-висока от най-високата оценка
-- на студентите от Sofia"

SELECT Name, Grade
FROM Students
WHERE Grade > (
    SELECT MAX(Grade)
    FROM Students
    WHERE City = 'Sofia'
);


-- Мислене:
--
-- Sofia
-- → филтрирам Sofia
-- → намирам MAX
-- → получавам една стойност
-- → сравнявам външните студенти


-- ============================================================
-- 20. ОСНОВНИЯТ РЕД НА МИСЛЕНЕ
-- ============================================================

-- НЕ започвай с:
--
-- "Какъв SQL да напиша?"
--
-- Започвай с:

-- 1. Какво ме пита задачата?
-- 2. Какви данни ми трябват?
-- 3. От една или от няколко таблици?
-- 4. Филтрирам ли РЕДОВЕ?
-- 5. Трябва ли да свържа таблици?
-- 6. Трябва ли да ГРУПИРАМ?
-- 7. Трябва ли да изчислявам COUNT/SUM/AVG/MIN/MAX?
-- 8. Филтрирам ли ГРУПИ?
-- 9. Трябва ли ми стойност от друга заявка?
-- 10. Какво точно трябва да покажа?


-- ============================================================
-- 21. МИСЛОВНА КАРТА
-- ============================================================

-- КОНКРЕТНИ РЕДОВЕ?
-- → WHERE

-- СВЪРЗВАМ ТАБЛИЦИ?
-- → JOIN + ON

-- ВСИЧКИ ОТ ЛЯВАТА ТАБЛИЦА?
-- → LEFT JOIN

-- ГРУПИРАМ?
-- → GROUP BY

-- БРОЯ?
-- → COUNT

-- СЪБИРАМ?
-- → SUM

-- СРЕДНО?
-- → AVG

-- НАЙ-МАЛКО?
-- → MIN

-- НАЙ-ГОЛЯМО?
-- → MAX

-- УСЛОВИЕ ВЪРХУ ГРУПА?
-- → HAVING

-- ТРЯБВА МИ РЕЗУЛТАТ ОТ ДРУГА ЗАЯВКА?
-- → SUBQUERY


-- ============================================================
-- 22. СИНТАКТИЧЕН "СКЕЛЕТ"
-- ============================================================

SELECT [какво показвам]
FROM [таблица]
JOIN [друга таблица]
    ON [връзка]
WHERE [условие за редове]
GROUP BY [по какво групирам]
HAVING [условие за групи]
ORDER BY [как сортирам];


-- Не всеки SELECT има всички части!
--
-- Използвам само това, което логиката на задачата изисква.


-- ============================================================
-- 23. НАЙ-ВАЖНОТО
-- ============================================================

-- УСЛОВИЕ
--     ↓
-- МИСЛЕНЕ
--     ↓
-- ЛОГИКА
--     ↓
-- ИНСТРУМЕНТ
--     ↓
-- СИНТАКСИС
--     ↓
-- КОД


-- Не е нужно да помня кода наизуст.
--
-- Ако знам:
-- "Трябва да филтрирам редове"
-- → WHERE
--
-- "Трябва да филтрирам групи"
-- → HAVING
--
-- "Трябва да намеря стойност и да я използвам другаде"
-- → SUBQUERY
--
-- "Данните са в две таблици"
-- → JOIN
--
-- "Искам отделен резултат за всеки клиент"
-- → GROUP BY
--
-- тогава мога да възстановя SQL-а.


-- ============================================================
-- ЗАПОМНИ КАТО ЛОГИКА, НЕ КАТО ШАБЛОН:
--
-- WHERE      → РЕДОВЕ
-- GROUP BY   → ГРУПИ
-- HAVING     → ГРУПИ
-- JOIN       → ВРЪЗКА
-- ON         → УСЛОВИЕ ЗА ВРЪЗКАТА
-- COUNT      → БРОЙ
-- SUM        → СУМА
-- AVG        → СРЕДНО
-- MIN        → НАЙ-МАЛКО
-- MAX        → НАЙ-ГОЛЯМО
-- SUBQUERY   → "ПЪРВО МИ ТРЯБВА ДРУГ РЕЗУЛТАТ"
-- ============================================================

Това бих ти го оставил като SQL карта за връщане назад. Най-важната част за теб не е да помниш целия блок, а да можеш да си кажеш:

„Какво ме пита задачата?“ → „Какво трябва да направя?“ → „Кой инструмент от картата ми трябва?“

И оттам синтаксисът вече се възстановява много по-лесно.
