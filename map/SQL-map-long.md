-- ============================================================
-- SQL — ЛОГИКА + ПОДРЕДБА НА СИНТАКСИСА
-- ============================================================


-- ============================================================
-- 1. ОСНОВЕН SELECT
-- ============================================================

SELECT [какво показвам]
FROM [откъде взимам данните]
WHERE [кои РЕДОВЕ оставям]
ORDER BY [как сортирам];


-- ЛОГИКА:
--
-- SELECT → какво показвам
-- FROM   → откъде
-- WHERE  → кои редове
-- ORDER BY → как ги подреждам


-- Пример:

SELECT Name, Grade
FROM Students
WHERE Grade > 50
ORDER BY Grade DESC;


-- ВАЖНО:
--
-- SELECT е написан първи,
-- но логически мислим:
--
-- FROM → WHERE → SELECT → ORDER BY
--
-- За учене обаче запомняме синтактичната подредба.


-- ============================================================
-- 2. TOP
-- ============================================================

SELECT TOP 3 Name, Grade
FROM Students
ORDER BY Grade DESC;


-- TOP е след SELECT.
--
-- Логика:
-- сортирам → взимам първите 3


-- ============================================================
-- 3. АГРЕГАТНА ФУНКЦИЯ
-- ============================================================

SELECT SUM(Price)
FROM Orders;


-- Мога да дам alias:

SELECT SUM(Price) AS TotalPrice
FROM Orders;


-- COUNT:

SELECT COUNT(*) AS OrderCount
FROM Orders;


-- AVG:

SELECT AVG(Grade) AS AverageGrade
FROM Students;


-- MIN / MAX:

SELECT MIN(Grade) AS MinGrade
FROM Students;

SELECT MAX(Grade) AS MaxGrade
FROM Students;


-- ============================================================
-- 4. GROUP BY
-- ============================================================

SELECT Customer, SUM(Price) AS TotalPrice
FROM Orders
GROUP BY Customer;


-- ПОДРЕДБА:
--
-- SELECT
-- FROM
-- GROUP BY
--
-- ЛОГИКА:
--
-- групирам по Customer
-- → SUM за всяка група
-- → показвам Customer + SUM


-- ============================================================
-- 5. GROUP BY + COUNT
-- ============================================================

SELECT Customer, COUNT(*) AS OrderCount
FROM Orders
GROUP BY Customer;


-- ЛОГИКА:
--
-- Customer → група
-- COUNT → броя във всяка група
-- SELECT → показвам групата + резултата


-- ============================================================
-- 6. GROUP BY + HAVING
-- ============================================================

SELECT Customer, SUM(Price) AS TotalPrice
FROM Orders
GROUP BY Customer
HAVING SUM(Price) > 500;


-- ПОДРЕДБАТА Е:
--
-- SELECT
-- FROM
-- GROUP BY
-- HAVING
--
-- НЕ:
--
-- GROUP BY
-- HAVING
-- WHERE
--
-- WHERE е ПРЕДИ GROUP BY.


-- ============================================================
-- 7. WHERE + GROUP BY + HAVING
-- ============================================================

SELECT Customer, SUM(Price) AS TotalPrice
FROM Orders
WHERE Price > 100
GROUP BY Customer
HAVING SUM(Price) > 500;


-- ЗАПОМНИ ПОДРЕДБАТА:
--
-- WHERE
-- ↓
-- редовете
--
-- GROUP BY
-- ↓
-- групите
--
-- HAVING
-- ↓
-- филтрирам групите


-- ЛОГИКА:
--
-- 1. Кои редове влизат?
-- 2. Как ги групирам?
-- 3. Какво изчислявам?
-- 4. Кои групи оставям?


-- ============================================================
-- 8. JOIN — ОСНОВЕН СИНТАКСИС
-- ============================================================

SELECT c.Name, o.Price
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId;


-- ПОДРЕДБА:
--
-- SELECT
-- FROM първа таблица
-- JOIN втора таблица
-- ON условие за връзката


-- ЗАПОМНИ:
--
-- FROM → откъде започвам
-- JOIN → коя таблица добавям
-- ON → по какво ги свързвам


-- ============================================================
-- 9. JOIN + WHERE
-- ============================================================

SELECT c.Name, o.Price
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
WHERE o.Price > 500;


-- ПОДРЕДБА:
--
-- FROM
-- JOIN
-- ON
-- WHERE


-- ЛОГИКА:
--
-- свързвам таблиците
-- → после филтрирам редовете


-- ============================================================
-- 10. JOIN + GROUP BY
-- ============================================================

SELECT c.Name, SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name;


-- ПОДРЕДБА:
--
-- SELECT
-- FROM
-- JOIN
-- ON
-- GROUP BY


-- ЛОГИКА:
--
-- връзка
-- → групирам по клиент
-- → SUM
-- → показвам клиента + SUM


-- ВАЖНО:
--
-- Ако показвам c.Name и използвам SUM,
-- обикновено трябва да имам c.Name
-- в GROUP BY.


-- ============================================================
-- 11. JOIN + GROUP BY + HAVING
-- ============================================================

SELECT c.Name, SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name
HAVING SUM(o.Price) > 500;


-- ПОДРЕДБА:
--
-- SELECT
-- FROM
-- JOIN
-- ON
-- GROUP BY
-- HAVING


-- ТОВА Е МНОГО ВАЖЕН ШАБЛОН:
--
-- свързвам
-- → групирам
-- → изчислявам
-- → филтрирам групата
-- → показвам


-- ============================================================
-- 12. JOIN + WHERE + GROUP BY + HAVING
-- ============================================================

SELECT c.Name, SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
WHERE o.Price > 100
GROUP BY c.Id, c.Name
HAVING SUM(o.Price) > 500;


-- ПОДРЕДБА:
--
-- SELECT
-- FROM
-- JOIN
-- ON
-- WHERE
-- GROUP BY
-- HAVING


-- ЛОГИКА:
--
-- JOIN
-- → свързвам
--
-- WHERE
-- → махам ненужните РЕДОВЕ
--
-- GROUP BY
-- → правя ГРУПИ
--
-- SUM
-- → изчислявам
--
-- HAVING
-- → филтрирам ГРУПИТЕ


-- ============================================================
-- 13. LEFT JOIN
-- ============================================================

SELECT c.Name, o.Price
FROM Customers AS c
LEFT JOIN Orders AS o
    ON c.Id = o.CustomerId;


-- ПОДРЕДБАТА Е СЪЩАТА:
--
-- FROM
-- LEFT JOIN
-- ON
--
-- ЛОГИКА:
--
-- "Искам всички от лявата таблица."


-- ============================================================
-- 14. LEFT JOIN + GROUP BY
-- ============================================================

SELECT c.Name, SUM(o.Price) AS TotalPrice
FROM Customers AS c
LEFT JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name;


-- ЛОГИКА:
--
-- запази всички клиенти
-- → свържи поръчките
-- → групирай по клиент
-- → SUM


-- ============================================================
-- 15. COUNT + GROUP BY + HAVING
-- ============================================================

SELECT c.Name, COUNT(o.Id) AS OrderCount
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name
HAVING COUNT(o.Id) >= 2;


-- ЛОГИКА:
--
-- клиент
-- → поръчки
-- → групирам
-- → COUNT
-- → оставям само COUNT >= 2


-- ============================================================
-- 16. COUNT + SUM + HAVING
-- ============================================================

SELECT
    c.Name,
    COUNT(o.Id) AS OrderCount,
    SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name
HAVING COUNT(o.Id) > 1
   AND SUM(o.Price) > 500;


-- ЛОГИКА:
--
-- група по клиент
-- → COUNT
-- → SUM
-- → двете условия са за ГРУПАТА
-- → HAVING


-- ============================================================
-- 17. SUBQUERY — ЕДНА СТОЙНОСТ
-- ============================================================

SELECT Name, Grade
FROM Students
WHERE Grade > (
    SELECT AVG(Grade)
    FROM Students
);


-- ПОДРЕДБА:
--
-- външен SELECT
-- WHERE
-- (
--     вътрешен SELECT
-- )
--
--
-- ЛОГИКА:
--
-- първо получавам една стойност
-- → после я използвам отвън


-- ============================================================
-- 18. SUBQUERY С MAX
-- ============================================================

SELECT Name, Grade
FROM Students
WHERE Grade = (
    SELECT MAX(Grade)
    FROM Students
);


-- Вътрешният SELECT:
--
-- SELECT MAX(Grade)
-- FROM Students
--
-- → връща една стойност
--
-- Външният:
--
-- WHERE Grade = тази стойност


-- ============================================================
-- 19. SUBQUERY С КОНКРЕТЕН ЧОВЕК
-- ============================================================

SELECT Name, Grade
FROM Students
WHERE Grade > (
    SELECT Grade
    FROM Students
    WHERE Name = 'Maria'
);


-- ПОДРЕДБА НА ВЪТРЕШНИЯ SELECT:
--
-- SELECT → какво търся
-- FROM   → откъде
-- WHERE  → кой конкретно


-- ЛОГИКА:
--
-- намирам Grade на Maria
-- → получавам една стойност
-- → сравнявам останалите


-- ============================================================
-- 20. SUBQUERY + WHERE ВЪТРЕ
-- ============================================================

SELECT Name, Grade
FROM Students
WHERE Grade > (
    SELECT MAX(Grade)
    FROM Students
    WHERE City = 'Sofia'
);


-- ВЪТРЕ:
--
-- FROM
-- → WHERE City = Sofia
-- → MAX
--
-- Резултат:
-- една стойност
--
-- ВЪН:
--
-- WHERE Grade > тази стойност


-- ============================================================
-- 21. SUBQUERY — КОГАТО МИ ТРЯБВА ДРУГ РЕЗУЛТАТ
-- ============================================================

-- Питай се:
--
-- "Мога ли първо да намеря нещо
-- и после да използвам намереното?"
--
-- ДА
-- ↓
-- SUBQUERY


-- ============================================================
-- 22. КОГА КАКВО ИЗПОЛЗВАМ
-- ============================================================

-- Искам конкретни РЕДОВЕ
-- → WHERE

-- Искам да свържа таблици
-- → JOIN + ON

-- Искам ВСИЧКИ от основната таблица
-- → LEFT JOIN

-- Искам ГРУПА за всеки клиент/град/etc.
-- → GROUP BY

-- Искам БРОЙ
-- → COUNT

-- Искам ОБЩА СУМА
-- → SUM

-- Искам СРЕДНА стойност
-- → AVG

-- Искам НАЙ-МАЛКА стойност
-- → MIN

-- Искам НАЙ-ГОЛЯМА стойност
-- → MAX

-- Искам условие върху ГРУПИ
-- → HAVING

-- Искам резултат от друга заявка
-- → SUBQUERY


-- ============================================================
-- 23. ОСНОВНА ПОДРЕДБА ЗА ЗАПОМНЯНЕ
-- ============================================================

SELECT
FROM
JOIN
ON
WHERE
GROUP BY
HAVING
ORDER BY;


-- НЕ ВСИЧКИ СА ЗАДЪЛЖИТЕЛНИ.
--
-- Използваш само тези, които задачата изисква.


-- Например:
--
-- Обикновен:
--
-- SELECT
-- FROM
-- WHERE
-- ORDER BY
--
--
-- JOIN:
--
-- SELECT
-- FROM
-- JOIN
-- ON
-- WHERE
-- ORDER BY
--
--
-- GROUP:
--
-- SELECT
-- FROM
-- WHERE
-- GROUP BY
-- HAVING
-- ORDER BY
--
--
-- JOIN + GROUP:
--
-- SELECT
-- FROM
-- JOIN
-- ON
-- WHERE
-- GROUP BY
-- HAVING
-- ORDER BY


-- ============================================================
-- 24. ВАЖНО: КАКВО МОЖЕ ДА ПОКАЗВАМ?
-- ============================================================

-- Ако имам:

SELECT Customer, SUM(Price)
FROM Orders
GROUP BY Customer;


-- Мога да покажа:
--
-- Customer → защото групирам по него
-- SUM(Price) → защото е агрегат


-- Не мога просто да добавя произволна колона:

SELECT Customer, Date, SUM(Price)
FROM Orders
GROUP BY Customer;


-- Защото Date не е:
--
-- 1. в GROUP BY
-- или
-- 2. агрегирана


-- ЗАПОМНИ:
--
-- При GROUP BY:
--
-- SELECT колоните трябва да са:
--
-- → в GROUP BY
-- ИЛИ
-- → агрегатна функция


-- ============================================================
-- 25. НАЙ-ВАЖНАТА МИНИ ШПАРГАЛКА
-- ============================================================

-- WHERE      → РЕДОВЕ
-- GROUP BY   → ГРУПИ
-- HAVING     → ГРУПИ
-- JOIN       → ВРЪЗКА
-- ON         → КАК СА СВЪРЗАНИ
-- COUNT      → БРОЙ
-- SUM        → СУМА
-- AVG        → СРЕДНО
-- MIN        → НАЙ-МАЛКО
-- MAX        → НАЙ-ГОЛЯМО
-- SUBQUERY   → ПЪРВО МИ ТРЯБВА ДРУГ РЕЗУЛТАТ


-- ============================================================
-- 26. СИНТАКСИСЪТ КАТО "СТЪЛБА"
-- ============================================================

-- SELECT
--    ↓
-- FROM
--    ↓
-- JOIN
--    ↓
-- ON
--    ↓
-- WHERE
--    ↓
-- GROUP BY
--    ↓
-- HAVING
--    ↓
-- ORDER BY


-- Не означава, че ВСЯКА заявка има всички стъпки.
--
-- Просто това е максималната стандартна подредба,
-- която да държиш в главата си.


-- ============================================================
-- 27. ЛОГИЧЕСКО ЧЕТЕНЕ НА ГОТОВ SQL
-- ============================================================

-- Когато видя:

SELECT ...
FROM ...
JOIN ...
ON ...
WHERE ...
GROUP BY ...
HAVING ...
ORDER BY ...


-- НЕ започвам да го чета като код.
--
-- Превеждам го:

-- Откъде?
-- ↓
-- Свързвам ли?
-- ↓
-- Кои редове оставям?
-- ↓
-- Как групирам?
-- ↓
-- Какво изчислявам?
-- ↓
-- Кои групи оставям?
-- ↓
-- Какво показвам?
-- ↓
-- Как сортирам?


-- ============================================================
-- 28. ФИНАЛЕН МОДЕЛ
-- ============================================================

-- УСЛОВИЕ
--    ↓
-- Какво ме питат?
--    ↓
-- Кои данни ми трябват?
--    ↓
-- Една или няколко таблици?
--    ↓
-- РЕДОВЕ или ГРУПИ?
--    ↓
-- Има ли агрегат?
--    ↓
-- Трябва ли JOIN?
--    ↓
-- Трябва ли SUBQUERY?
--    ↓
-- Подреждам синтаксиса
--    ↓
-- Пиша SQL
--    ↓
-- Проверявам резултата


-- ============================================================
-- НЕ ЗУБРИМ:
--
-- SELECT ... FROM ... WHERE ...
--
-- А РАЗБИРАМЕ:
--
-- "Искам тези данни"
-- → SELECT
--
-- "От тази таблица"
-- → FROM
--
-- "Само тези редове"
-- → WHERE
--
-- "Тези таблици са свързани"
-- → JOIN + ON
--
-- "Искам един резултат за всеки клиент"
-- → GROUP BY
--
-- "Искам да броя/събирам"
-- → COUNT / SUM
--
-- "Искам да махна някои групи"
-- → HAVING
--
-- "Първо ми трябва резултат от друга заявка"
-- → SUBQUERY
-- ============================================================
Това вече е много по-близо до начина, по който ти реално учиш: не просто „какво прави JOIN“, а къде стои JOIN, къде стои ON, кога идва WHERE, кога GROUP BY, кога HAVING и какво мога да използвам след като съм създал групите.

Особено важно е да запомниш тази стълба:

SELECT
FROM
JOIN
ON
WHERE
GROUP BY
HAVING
ORDER BY

Не като задължителен шаблон, а като реда, в който SQL конструкциите се подреждат. После задачата ти казва кои от тях изобщо са нужни.
