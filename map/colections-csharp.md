/*
Array
+ бърз индекс
+ прост
- фиксиран размер


List<T>
+ лесен
+ динамичен
+ индекс
+ много удобен за LINQ
- lookup по стойност не е основната му сила


Dictionary<TKey,TValue>
+ бързо търсене по key
+ идеален за ID → object
- key трябва да е уникален

HashSet<T>
+ уникалност
+ бърз Contains

- няма key → value


Queue<T>
+ FIFO
+ идеален за опашки
- не е за произволен достъп


Stack<T>
+ LIFO
+ история / undo / nested operations
- не е за произволен достъп


LinkedList<T>
+ специфични операции около node
- по-сложен
- няма бърз index

    .ToList();

// Where
// ↓
// OrderBy
// ↓

// ↓
// ToList

// НЕ:

//
// "Как ще използвам данните?"

//
// Първо МИСЛЯ за начина на достъп.
