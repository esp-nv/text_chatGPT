-- ============================================================
-- SQL SERVER — ШПАРГАЛКА
-- ============================================================


-- ============================================================
-- 1. DISTINCT
-- ============================================================

-- ЦЕЛ:
-- → уникални резултати
-- → маха повторенията

SELECT DISTINCT City
FROM Students;

-- ЛОГИКА:
-- DISTINCT → "Не искам дубликати."


-- ============================================================
-- 2. INDEX
-- ============================================================

-- ЦЕЛ:
-- → ускорява търсенето

CREATE INDEX IX_Students_Name
ON Students(Name);

-- После:

SELECT *
FROM Students
WHERE Name = 'Maria';

-- ЛОГИКА:
-- INDEX → "Търсенето по тази колона да е по-бързо."

-- ВНИМАНИЕ:
-- → заема място
-- → INSERT / UPDATE / DELETE могат да станат по-тежки


-- ============================================================
-- 3. FUNCTION
-- ============================================================

-- ЦЕЛ:
-- → обработвам нещо
-- → получавам резултат

-- Примери за готови SQL Server функции:

SELECT LEN(Name)
FROM Students;

SELECT GETDATE();

SELECT SUM(Price)
FROM Orders;

SELECT AVG(Grade)
FROM Students;

-- ЛОГИКА:
-- FUNCTION → "Дай ми резултат от тази операция."


-- ============================================================
-- 4. STORED PROCEDURE
-- ============================================================

-- ЦЕЛ:
-- → готова SQL операция
-- → извиквам я

-- Пример:

CREATE PROCEDURE GetCustomerOrders
    @CustomerId INT
AS
BEGIN
    SELECT *
    FROM Orders
    WHERE CustomerId = @CustomerId;
END;


-- Извикване:

EXEC GetCustomerOrders @CustomerId = 5;


-- ЛОГИКА:
-- STORED PROCEDURE
-- → "Имам готова операция → извикай я."


-- ============================================================
-- 5. TRIGGER
-- ============================================================

-- ЦЕЛ:
-- → случва се събитие
-- → автоматично се изпълнява действие

-- Пример:
-- При изтриване на Order записваме информация в Audit таблица.

CREATE TRIGGER trg_OrderDelete
ON Orders
AFTER DELETE
AS
BEGIN
    INSERT INTO OrderAudit(OrderId, DeletedAt)
    SELECT Id, GETDATE()
    FROM deleted;
END;


-- ЛОГИКА:
--
-- DELETE
--    ↓
-- TRIGGER
--    ↓
-- автоматично действие


-- ЗАПОМНИ:
-- TRIGGER → "Стана това → автоматично направи онова."


-- ============================================================
-- 6. DELETE — ИЗТРИВАНЕ
-- ============================================================

-- Изтривам конкретен ред:

DELETE FROM Students
WHERE Id = 5;


-- !!! ВНИМАНИЕ !!!
-- Ако няма WHERE:

DELETE FROM Students;

-- → изтривам ВСИЧКИ редове от таблицата.


-- ЛОГИКА:
-- DELETE → "Кои редове искам да изтрия?"
-- WHERE  → "Точно тези."


-- ============================================================
-- 7. TRANSACTION
-- ============================================================

-- ЦЕЛ:
-- → няколко операции да се третират като едно цяло

BEGIN TRANSACTION;

UPDATE Accounts
SET Balance = Balance - 100
WHERE Id = 1;

UPDATE Accounts
SET Balance = Balance + 100
WHERE Id = 2;

COMMIT;


-- ЛОГИКА:
--
-- BEGIN TRANSACTION
-- → започвам групата операции
--
-- операции
-- → правя промените
--
-- COMMIT
-- → всичко е наред → запази


-- ============================================================
-- 8. ROLLBACK
-- ============================================================

BEGIN TRANSACTION;

DELETE FROM Students
WHERE Id = 5;

ROLLBACK;


-- Резултат:
-- DELETE се отменя.


-- ЛОГИКА:
--
-- BEGIN TRANSACTION
--       ↓
-- DELETE
--       ↓
-- "Не искам тази промяна"
--       ↓
-- ROLLBACK
--       ↓
-- връщам назад


-- ============================================================
-- 9. COMMIT
-- ============================================================

BEGIN TRANSACTION;

DELETE FROM Students
WHERE Id = 5;

COMMIT;


-- Резултат:
-- DELETE остава.


-- ЛОГИКА:
--
-- COMMIT → "Потвърждавам промените."


-- ============================================================
-- 10. DELETE + TRANSACTION — НАЙ-ВАЖНИЯТ ПРИМЕР
-- ============================================================

BEGIN TRANSACTION;

DELETE FROM Orders
WHERE CustomerId = 5;

-- Проверявам какво съм направил:

SELECT *
FROM Orders
WHERE CustomerId = 5;

-- Ако НЕ съм доволен:

ROLLBACK;


-- Ако всичко е правилно:

-- COMMIT;


-- ЛОГИКА:
--
-- BEGIN
-- → започвам
--
-- DELETE
-- → правя промяна
--
-- SELECT
-- → проверявам
--
-- ROLLBACK
-- → отменям
--
-- COMMIT
-- → потвърждавам


-- ============================================================
-- 11. VIEW
-- ============================================================

-- ЦЕЛ:
-- → запазвам сложна SELECT заявка
-- → после я използвам като виртуална таблица

CREATE VIEW CustomerTotals AS
SELECT
    c.Id,
    c.Name,
    SUM(o.Price) AS TotalPrice
FROM Customers AS c
JOIN Orders AS o
    ON c.Id = o.CustomerId
GROUP BY c.Id, c.Name;


-- После:

SELECT *
FROM CustomerTotals;


-- И мога да филтрирам:

SELECT *
FROM CustomerTotals
WHERE TotalPrice > 500;


-- ЛОГИКА:
--
-- сложна SELECT заявка
--        ↓
--       VIEW
--        ↓
-- използвам я като виртуална таблица


-- VIEW ≠ физическа таблица
-- VIEW → представя резултат от заявка


-- ============================================================
-- 12. FUNCTION vs PROCEDURE vs VIEW vs TRIGGER
-- ============================================================

-- FUNCTION
-- → изчислявам / обработвам
-- → получавам резултат


-- STORED PROCEDURE
-- → готова операция
-- → EXEC


-- VIEW
-- → запазена SELECT заявка
-- → използвам я като виртуална таблица


-- TRIGGER
-- → събитие
-- → автоматично действие


-- ============================================================
-- 13. TRANSACTION — МИСЛОВЕН МОДЕЛ
-- ============================================================

BEGIN TRANSACTION
        ↓
     операции
        ↓
   всичко ОК?
      /    \
    ДА      НЕ
    ↓        ↓
 COMMIT   ROLLBACK


-- COMMIT
-- → ЗАПАЗИ


-- ROLLBACK
-- → ВЪРНИ НАЗАД


-- ============================================================
-- 14. ЗАПОМНИ КАТО ЛОГИКА
-- ============================================================

-- DISTINCT
-- → УНИКАЛНИ РЕЗУЛТАТИ


-- INDEX
-- → ПО-БЪРЗО ТЪРСЕНЕ


-- FUNCTION
-- → ОБРАБОТКА → РЕЗУЛТАТ


-- STORED PROCEDURE
-- → ГОТОВА ОПЕРАЦИЯ → EXEC


-- TRIGGER
-- → СЪБИТИЕ → АВТОМАТИЧНО ДЕЙСТВИЕ


-- DELETE
-- → ИЗТРИВАМ РЕДОВЕ


-- TRANSACTION
-- → ГРУПИРАМ ОПЕРАЦИИ


-- COMMIT
-- → ПОТВЪРЖДАВАМ / ЗАПАЗВАМ


-- ROLLBACK
-- → ОТМЕНЯМ / ВРЪЩАМ


-- VIEW
-- → ЗАПАЗЕНА ЗАЯВКА → ВИРТУАЛНА ТАБЛИЦА


-- ============================================================
-- 15. МИНИ ШПАРГАЛКА
-- ============================================================

-- DISTINCT  → уникални
-- INDEX     → бързо търсене
-- FUNCTION  → резултат
-- PROCEDURE → готова операция
-- TRIGGER   → автоматично действие
-- DELETE    → изтриване
-- TRANSACTION → група операции
-- COMMIT    → запази
-- ROLLBACK  → върни
-- VIEW      → виртуална таблица


-- ============================================================
-- НАЙ-ВАЖНОТО:
--
-- Не учим:
-- "CREATE INDEX има точно този синтаксис..."
--
-- Учим:
--
-- INDEX?
-- → защо ми трябва?
-- → по-бързо търсене
--
-- TRANSACTION?
-- → защо?
-- → няколко операции като едно цяло
--
-- ROLLBACK?
-- → защо?
-- → нещо се обърка → връщам назад
--
-- TRIGGER?
-- → защо?
-- → събитие → автоматично действие
--
-- VIEW?
-- → защо?
-- → искам сложната заявка да мога да използвам
--   като виртуална таблица
--
-- ТОВА Е ЛОГИКАТА.
-- СИНТАКСИСЪТ МОЖЕ ДА СЕ ПРОВЕРИ.
